import os import base64 from cryptography.fernet import Fernet # 1. Извлекаем Hex-ключ из Secrets Replit hex_key = os.getenv("ENCRYPTION_KEY") if not hex_key: # Фоллбэк-заглушка на случай, если ключ забыли прописать в Secrets (для предотвращения падения) print("WARNING: ENCRYPTION_KEY is not set in Secrets! Using temporary fallback.") # Используем ваш старый SESSION_SECRET как соль для генерации временного ключа session_secret = os.getenv("SESSION_SECRET", "default_fallback_secret_32_bytes_long_!!") # Переводим в 32 байта для Fernet fernet_key = base64.urlsafe_b64encode(session_secret[:32].encode().ljust(32, b'_')) else: # 2. Правильная конвертация 64-hex символов (32 байта) в URL-safe Base64 для Fernet raw_bytes = bytes.fromhex(hex_key) fernet_key = base64.urlsafe_b64encode(raw_bytes) # Инициализируем объект шифрования fernet = Fernet(fernet_key) # Теперь функции fernet.encrypt() и fernet.decrypt() используют выделенный 256-битный ключ
