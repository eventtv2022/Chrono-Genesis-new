2. GOOGLE SHEETS PERMISSION SCRIPT (TypeScript)
Попроси агента создать и запустить скрипт scripts/grant_access.ts, который использует учетные данные сервисного аккаунта для выдачи вам прав:

TypeScript
// scripts/grant_access.ts
import { google } from 'googleapis';
import { getServiceAccountAuth } from '../lib/google/auth'; // Твой путь к auth

async function grantAccess() {
  const auth = await getServiceAccountAuth();
  const drive = google.drive({ version: 'v3', auth });
  
  await drive.permissions.create({
    fileId: '13PgzgxjZ60cZT5ATbDYyTLi01psHIYxRoTOJY-CWNw4',
    requestBody: {
      role: 'writer',
      type: 'user',
      emailAddress: 'ВАШ_EMAIL@gmail.com', // <--- Вставь сюда свою почту
    },
  });
  console.log("Доступ успешно предоставлен!");
}
grantAccess();

        citizen.balance_minutes
      ]],
    },
  });
}
