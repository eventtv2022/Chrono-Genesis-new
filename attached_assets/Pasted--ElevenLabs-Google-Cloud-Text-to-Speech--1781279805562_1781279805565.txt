Реализуй обработчик формы, который принимает данные, запускает ИИ для «развития мысли» Гражданина, а затем использует **Google Text-to-Speech** для генерации аудиофайла.

```python
import os
from fastapi import APIRouter, Depends, UploadFile, File, Form
from google.cloud import texttospeech
from sqlalchemy.orm import Session

router = APIRouter()

# Инициализация клиента Google TTS

tts_client = texttospeech.TextToSpeechClient()

    birth_date: str = Form(...),
    email: str = Form(...),
    short_thought: str = Form(...), # Исходная мысль гражданина
    passport_photo: UploadFile = File(...),
    db: Session = Depends(get_db)
):
    # 1. Сохраняем фото в локальное хранилище assets
    photo_dir = "storage/avatars"
    os.makedirs(photo_dir, exist_ok=True)
    file_path = f"{photo_dir}/{surname}_{email}.jpg"
    with open(file_path, "wb") as buffer:
        buffer.write(await passport_photo.read())

    # 2. ПОДКАПОТНЫЙ ИИ: Развиваем короткую мысль гражданина в 90-секундный манифест
    # (Вызов вашей LLM с системной инструкцией не копировать фразу, а масштабировать её)
    expanded_manifesto = expand_thought_via_llm(short_thought)

    # 3. GOOGLE TEXT-TO-SPEECH ENGINE
    # Настраиваем текст для озвучки
    synthesis_input = texttospeech.SynthesisInput(text=expanded_manifesto)

    # Настройки голоса (Выбираем качественный русский Wavenet или Neural2 голос)
    voice = texttospeech.VoiceSelectionParams(
        language_code="ru-RU",
        name="ru-RU-Wavenet-D", # Глубокий мужской голос (или ru-RU-Neural2-B)
        ssml_gender=texttospeech.SsmlVoiceGender.MALE
    )

    # Настройки аудио-контейнера (MP3 высокого качества)
    audio_config = texttospeech.AudioConfig(
        audio_encoding=texttospeech.AudioEncoding.MP3,
        speaking_rate=1.0, # Естественная скорость
        pitch=0.0
    )

    # Синтез речи
    response = tts_client.synthesize_speech(
        input=synthesis_input, voice=voice, audio_config=audio_config
    )

    # Сохраняем аудиодорожку для капсулы
    audio_output_path = f"storage/audio/{surname}_{email}.mp3"
    os.makedirs("storage/audio", exist_ok=True)
    with open(audio_output_path, "wb") as out:
        out.write(response.audio_content)

        email=email,

        ai_manifesto=expanded_manifesto,
        audio_path=audio_output_path
    )
    db.add(new_citizen)
    db.commit()

    return {
        "status": "success", 
        "message": "Гражданин зарегистрирован, мысль развита ИИ и озвучена через Google TTS"
```
