        let activeHour = 1;

        function initClock() {
            const container = document.getElementById('hours-container');
            const hand = document.getElementById('clock-hand');
            const radius = 130; // Радиус позиционирования точек
            const centerX = 170; // Координаты центра циферблата
            const centerY = 170;

            // Генерируем 24 часовые кнопки
            LANGUAGES_24.forEach((item) => {
                // Высчитываем тригонометрические координаты для каждого часа
                const angle = (item.hour * 15 * Math.PI) / 180 - Math.PI / 2;
                const x = centerX + radius * Math.cos(angle);
                const y = centerY + radius * Math.sin(angle);

                const btn = document.createElement('button');
                btn.id = `hour-${item.hour}`;
                btn.innerText = item.hour;
                btn.className = "absolute w-6 h-6 rounded-full bg-zinc-800 text-[10px] font-bold text-zinc-300 border border-zinc-700 hover:border-amber-500 hover:text-white flex items-center justify-center transition-all duration-200 z-10 cursor-pointer";
                btn.style.left = `${x - 12}px`; // Центрирование
                btn.style.top = `${y - 12}px`;

                btn.addEventListener('click', () => changeHour(item.hour));
                container.appendChild(btn);
            });

            // Показываем начальный статус (Русский, 1:00)
            changeHour(1);
        }

        function changeHour(hour) {
            // Снимаем класс активности с прошлой кнопки
            document.getElementById(`hour-${activeHour}`)?.classList.remove('active-hour');
            
            activeHour = hour;
            const item = LANGUAGES_24.find(l => l.hour === hour);
            
            // Включаем активную кнопку
            const activeBtn = document.getElementById(`hour-${hour}`);
            if (activeBtn) activeBtn.classList.add('active-hour');

            // Поворачиваем стрелку часов (360 градусов / 24 деления = 15 градусов на час)
            const hand = document.getElementById('clock-hand');
            hand.style.transform = `rotate(${hour * 15}deg)`;

            // Подставляем переводы
            const t = translations[item.code] || translations['ru'];
            document.getElementById('app-welcome').innerText = t.welcome;
            document.getElementById('app-select-hour').innerText = t.select;
            document.getElementById('app-status-text').innerText = t.status;
            document.getElementById('current-lang-name').innerText = `${item.label} (${hour}:00)`;

            // Поддержка направления письма справа налево (RTL) для Иврита, Арабского, Урду и Фарси
            if (item.rtl) {
                document.body.dir = 'rtl';
                document.body.style.textAlign = 'right';
            } else {
                document.body.dir = 'ltr';
                document.body.style.textAlign = 'center';
            }
        }

        // Запуск
        document.addEventListener('DOMContentLoaded', initClock);
